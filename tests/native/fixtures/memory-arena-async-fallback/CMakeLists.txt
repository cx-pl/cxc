cmake_minimum_required(VERSION 3.31)
project(cx_memory_arena_async_fallback C)
if(NOT DEFINED CX_GENERATED_DIR OR NOT DEFINED CXCORE_SOURCE_DIR)
    message(FATAL_ERROR "Set CX_GENERATED_DIR to compiler output and CXCORE_SOURCE_DIR to cxcore")
endif()
option(CX_STATIC_LINK "Use one statically linked runtime" ON)
set(CX_BUILD_STATIC ${CX_STATIC_LINK} CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${CXCORE_SOURCE_DIR} cxcore)
add_executable(memoryArenaAsyncFallback ${CX_GENERATED_DIR}/memoryArenaAsyncFallback.c)
target_include_directories(memoryArenaAsyncFallback PRIVATE ${CX_GENERATED_DIR})
target_link_libraries(memoryArenaAsyncFallback PRIVATE cxcore)
target_compile_features(memoryArenaAsyncFallback PRIVATE c_std_11)
target_compile_definitions(memoryArenaAsyncFallback PRIVATE CX_MEMORYARENAASYNCFALLBACK_BUILD)
if(NOT CX_STATIC_LINK)
    target_compile_definitions(memoryArenaAsyncFallback PRIVATE CX_DYNAMIC_MODULE)
    if(WIN32)
        add_custom_command(TARGET memoryArenaAsyncFallback POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:cxcore> $<TARGET_FILE_DIR:memoryArenaAsyncFallback>)
    endif()
endif()
enable_testing()
add_test(NAME memoryArenaAsyncFallback COMMAND memoryArenaAsyncFallback)
set_tests_properties(memoryArenaAsyncFallback PROPERTIES TIMEOUT 45)
